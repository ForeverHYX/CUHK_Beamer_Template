\section{Layout}

\begin{frame}{Page Styles}{LAYOUT STYLES}
  \vspace{1mm}
  \begin{minipage}[t]{0.48\textwidth}
    \vspace{0pt}%
    \begin{cuhkbox}{blue}{Auto-generated layouts}
      {\small
      \begin{denseitem}
        \item \textbf{Cover}: purple emblem \& wordmark + lavender title band
        \item \textbf{Section pages}: inserted by \texttt{\string\section}
        \item \textbf{Thanks page}: one line, \texttt{\string\cuhkthanks}
        \item \textbf{Header}: title + gradient rule + gradient wordmark
      \end{denseitem}}
    \end{cuhkbox}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.48\textwidth}
    \vspace{0pt}%
    \begin{cuhkbox}{green}{Frame titles \& subtitles}
      {\small Every frame title takes an optional subtitle:}

      \vspace{0.8mm}
      {\small\color{black!60}
      \texttt{\string\begin\{frame\}\{Title\}\{SUBTITLE\}}}

      \vspace{0.8mm}
      {\small The subtitle appears as small grey text next to
      the title; uppercase acronyms look best.}
    \end{cuhkbox}
  \end{minipage}
  \vspace{1.2mm}
  \begin{cuhkbox}{gray}{Pure white content pages}
    {\small Content pages carry no background decoration --- ideal for dense
    academic material; the cover and section pages are full-bleed
    designs with generous whitespace, on a 16:9 canvas.}
  \end{cuhkbox}
  \begin{takeaway}
    This page shows the header style: title + gradient rule + gradient wordmark.
  \end{takeaway}
\end{frame}

\begin{frame}{Multi-Column Layouts}{MULTI-COLUMN}
  % ---- two columns ----
  \begin{minipage}[t]{0.475\textwidth}
    \vspace{0pt}%
    \begin{cuhkbox}{blue}{Two columns: minipage + hfill}
      {\footnotesize Place content side by side with
      \texttt{minipage[t]}; start each column with
      \texttt{\string\vspace\{0pt\}} for top alignment and
      separate columns with \texttt{\string\hfill}.}
    \end{cuhkbox}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.475\textwidth}
    \vspace{0pt}%
    \begin{cuhkbox}{sky}{Any column ratio}
      {\footnotesize Widths combine freely:
      0.475 + 0.475, 0.3 + 0.3 + 0.3, 0.6 + 0.38, ...
      and mix \texttt{cuhkbox}, \texttt{imgcard}, charts, etc.}
    \end{cuhkbox}
  \end{minipage}
  \vspace{1.2mm}

  % ---- three columns ----
  \begin{minipage}[t]{0.31\textwidth}
    \vspace{0pt}%
    \begin{cuhkbox}{orange}{Column one}
      {\footnotesize For three columns, 0.31 + 0.31 + 0.31 works well.}
    \end{cuhkbox}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.31\textwidth}
    \vspace{0pt}%
    \begin{cuhkbox}{purple}{Column two}
      {\footnotesize Start every column with
      \texttt{\string\vspace\{0pt\}}.}
    \end{cuhkbox}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.31\textwidth}
    \vspace{0pt}%
    \begin{cuhkbox}{steel}{Column three}
      {\footnotesize Unequal heights align at the top automatically.}
    \end{cuhkbox}
  \end{minipage}

  \begin{takeaway}
    Remember: put \texttt{\string\vspace\{0pt\}} before the first element of every
    \texttt{minipage[t]}, otherwise tops will not align.
  \end{takeaway}
\end{frame}
